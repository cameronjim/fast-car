cmake_minimum_required(VERSION 3.8)
project(racer_control)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
endif()
if(CMAKE_COMPILER_IS_GNUCXX OR CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  add_compile_options(-Wall -Wextra)
endif()

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(ackermann_msgs REQUIRED)

# -- ROS-free core library (claude-docs/10-conventions.md: gate/decision logic separated
#    from node plumbing) -------------------------------------------------------------------
add_library(racer_control_core
  src/raceline.cpp
  src/pure_pursuit.cpp
)
target_include_directories(racer_control_core PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
)

# -- Generated vehicle_params bindings (claude-docs/06-vehicle-params.md rule 3: bindings
#    are GENERATED by tools/gen_params.py, never hand-written; claude-docs/10-conventions.md
#    / .gitignore: never committed -- CI/the build regenerates them). Only tracker_node.cpp
#    includes the generated header; racer_control_core (and its gtest suite) never does, so
#    the ROS-free core can be built and tested with no bindings generation step at all. `uv`
#    (already required repo tooling, claude-docs/00-project-overview.md "Quick facts") runs
#    tools/gen_params.py inside tools/'s own pinned venv (tools/pyproject.toml), so this
#    works identically on a dev machine and in the ros-dev CI image without needing
#    pyyaml/jsonschema installed into whatever python3 colcon happens to invoke.
get_filename_component(RACER_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(RACER_GENERATED_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
find_program(RACER_UV_EXECUTABLE uv REQUIRED)
add_custom_command(
  OUTPUT "${RACER_GENERATED_DIR}/vehicle_params_generated.hpp"
  COMMAND "${RACER_UV_EXECUTABLE}" run --project "${RACER_REPO_ROOT}/tools" python
          "${RACER_REPO_ROOT}/tools/gen_params.py"
          --params "${RACER_REPO_ROOT}/config/vehicle_params.yaml"
          --schema "${RACER_REPO_ROOT}/config/vehicle_params.schema.json"
          --out-dir "${RACER_GENERATED_DIR}"
  DEPENDS
    "${RACER_REPO_ROOT}/config/vehicle_params.yaml"
    "${RACER_REPO_ROOT}/config/vehicle_params.schema.json"
    "${RACER_REPO_ROOT}/tools/gen_params.py"
  COMMENT "racer_control: generating vehicle_params C++ binding (never committed, see claude-docs/06-vehicle-params.md)"
  VERBATIM
)
add_custom_target(racer_control_generate_vehicle_params
  DEPENDS "${RACER_GENERATED_DIR}/vehicle_params_generated.hpp"
)

# -- tracker_node executable --------------------------------------------------------------
add_executable(tracker_node src/tracker_node.cpp)
add_dependencies(tracker_node racer_control_generate_vehicle_params)
target_include_directories(tracker_node PRIVATE "${RACER_GENERATED_DIR}")
target_link_libraries(tracker_node racer_control_core)
ament_target_dependencies(tracker_node rclcpp rcl_interfaces nav_msgs ackermann_msgs)

install(TARGETS tracker_node
  DESTINATION lib/${PROJECT_NAME}
)

# -- pure_pursuit_cli: tiny CLI harness for the S.3 cross-language base-controller
#    divergence test (see src/pure_pursuit_cli.cpp's header comment). Built unconditionally
#    (not gated behind BUILD_TESTING) -- it is a small, useful-on-its-own diagnostic binary,
#    same treatment as tracker_node above, not test-only scaffolding. -------------------
add_executable(pure_pursuit_cli src/pure_pursuit_cli.cpp)
add_dependencies(pure_pursuit_cli racer_control_generate_vehicle_params)
target_include_directories(pure_pursuit_cli PRIVATE "${RACER_GENERATED_DIR}")
target_link_libraries(pure_pursuit_cli racer_control_core)

install(TARGETS pure_pursuit_cli
  DESTINATION lib/${PROJECT_NAME}
)

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/launch")
  install(DIRECTORY launch
    DESTINATION share/${PROJECT_NAME}
  )
endif()

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_racer_control_core
    test/test_pure_pursuit.cpp
    test/test_raceline.cpp
  )
  target_link_libraries(test_racer_control_core racer_control_core)
  target_compile_definitions(test_racer_control_core PRIVATE
    RACER_CONTROL_TEST_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures"
  )

  find_package(launch_testing_ament_cmake REQUIRED)
  add_launch_test(test/test_tracker_node_launch.py
    TARGET test_tracker_node_launch
  )

  # -- S.3 cross-language base-controller divergence test ------------------------------
  # Runs the committed fixture (training/racer_train/tests/fixtures/divergence_states.csv +
  # divergence_expected.json, generated deterministically by
  # training/racer_train/tests/generate_divergence_fixture.py from
  # racer_train.raceline.PurePursuitController, the Python port of this exact algorithm)
  # through the pure_pursuit_cli binary just built above, and asserts they agree within a
  # stated tolerance -- claude-docs/12-testing.md's L5 "envelope-in-env test" divergence
  # pattern, applied to the base controller. A plain CTest (not ament_cmake_pytest, which
  # this package does not otherwise need): compare_divergence.py is invoked directly as a
  # subprocess, and `colcon test` runs every CTest registered for an ament_cmake package.
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(
    NAME test_pure_pursuit_cli_divergence
    COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/test/divergence/compare_divergence.py"
            --cli "$<TARGET_FILE:pure_pursuit_cli>"
            --raceline "${RACER_REPO_ROOT}/config/tracks/gym_oval/raceline.csv"
            --states "${RACER_REPO_ROOT}/training/racer_train/tests/fixtures/divergence_states.csv"
            --expected "${RACER_REPO_ROOT}/training/racer_train/tests/fixtures/divergence_expected.json"
            --tolerance 1e-6
  )
  set_tests_properties(test_pure_pursuit_cli_divergence PROPERTIES
    LABELS "racer_control_divergence"
  )
endif()

ament_package()
